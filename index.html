# Create Simulator

set ns [new Simulator]

# Open Trace file and NAM file

set ntrace [open demo.tr w]

set namfile [open demo.nam w]

# Procedure to finish simulation and analyze results

proc Finish {} {

global ns ntrace namfile

# Dump all the trace data and close the files

$ns flush-trace

close $ntrace

close $namfile

# Execute the nam animation file (optional)

exec nam demo.nam &

# Show the number of packets dropped

set num_drops [exec grep -c "^d" demo.tr]

puts "The number of packet drops is $num_drops"

exit 0 }

# Trace all events to the trace file
$ns trace-all $ntrace
$ns namtrace-all $namfile

# Create 3 nodes
set n0 [$ns node]
set n1 [$ns node]
set n2 [$ns node]

# Label the nodes
$n0 label "TCP Source"
$n2 label "Sink"

# Set the color (optional)
$ns color 1 pink

# Create Links between nodes
# You can modify the bandwidth to observe variation in packet drops
$ns duplex-link $n0 $n1 1Mb 10ms DropTail
$ns duplex-link $n1 $n2 1Mb 10ms DropTail

# Make the Link Orientation
$ns duplex-link-op $n0 $n1 orient right
$ns duplex-link-op $n1 $n2 orient right

# Set Queue Size

# You can modify the queue length as well to observe variation in packet drops

$ns queue-limit $n0 $n1 10

$ns queue-limit $n1 $n2 10

# Set up a Transport layer connection.

set tcp0 [new Agent/TCP]

$ns attach-agent $n0 $tcp0

set sink0 [new Agent/TCPSink]

$ns attach-agent $n2 $sink0

$ns connect $tcp0 $sink0

# Set up an Application layer Traffic

set ftp0 [new Application/FTP]

$ftp0 set packetSize_ 100

$ftp0 set rate_ 1Mb

$ftp0 attach-agent $tcp0

$ftp0 set random_ false

# Schedule Events

$ns at 0.0 "$ftp0 start"

$ns at 5.0 "Finish"

# Run the Simulation

$ns run
